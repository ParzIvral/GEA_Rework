# Indochina presets: an explicit Heye route takes precedence over the no-revolt preset.
GEA_indochina_revolt_preset = {
	OR = {
		has_game_rule = { rule = GEA_future option = GEA_FUTURE_REVOLT_YES }
		has_game_rule = { rule = GEA_governor option = GEA_GOVERNOR_HEYE }
	}
}

GEA_indochina_no_revolt_preset = {
	has_game_rule = { rule = GEA_future option = GEA_FUTURE_REVOLT_NO }
	NOT = { has_game_rule = { rule = GEA_governor option = GEA_GOVERNOR_HEYE } }
}

GEA_indochina_random_preset = {
	has_game_rule = { rule = GEA_future option = DEFAULT }
	NOT = { has_game_rule = { rule = GEA_governor option = GEA_GOVERNOR_HEYE } }
}
