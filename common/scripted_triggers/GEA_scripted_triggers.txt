		NOT = {
			OR = {
				has_country_flag = GEA_china_warlord_interaction_blocked #禁止互动

			}
		}
		OR = {
			original_tag = LEP #德国人的狗
			AND = {
				original_tag = GXC
				has_country_leader = { name = "Chen Lianbo" ruling_only = yes } #老霸道了
			}
			AND = {
				original_tag = YUN
				has_country_leader = { name = "Tang Jiyao" ruling_only = yes } #弱智大唐
			}
			AND = {
				original_tag = QIE
				has_country_flag = wu_intervenes_proLEP #正式支持南京
			}
			AND = {
				original_tag = SZC
				NOT = { has_country_leader = { name = "Chen Jiongming" ruling_only = yes } } #联省逼
				NOT = { has_country_leader = { name = "Lü Chao" ruling_only = yes } } #吕超
				NOT = { has_completed_focus = SZC_Eradicate_The_Concessions }
				OR = {
					has_country_leader = { name = "Yang Sen" ruling_only = yes } #杨森
					has_country_leader = { name = "Liu Cunhou" ruling_only = yes } #留存后
					has_completed_focus = SZC_Contact_An_Old_Ally
				}
			}
		}
	}

GEA_east_china_coup_threshold_trigger = {#可政变条件
	OR = {
		AND = {
			FROM = { NOT = { has_variable = GEA_coup_capital_requirement } }
			check_variable = { var = GEA_china_lep_capital_influence value = 90 compare = greater_than_or_equals }
			check_variable = { var = GEA_china_lep_military_projection value = 80 compare = greater_than_or_equals }
		}
		AND = {
			FROM = { has_variable = GEA_coup_capital_requirement }
			check_variable = { var = GEA_china_lep_capital_influence value = FROM.GEA_coup_capital_requirement compare = greater_than_or_equals }
			check_variable = { var = GEA_china_lep_military_projection value = FROM.GEA_coup_military_requirement compare = greater_than_or_equals }
		}
	}
}
