GEA_ensure_german_loan_repayment_effect = {
	if = {
		limit = {
			has_country_flag = GEA_german_loan_active
			NOT = { has_variable = GEA_german_loan_repayment }
		}
		if = {
			limit = { has_active_mission = GEA_warlord_repay_german_loan_short }
			set_variable = { GEA_german_loan_repayment = 50 }
		}
		else_if = {
			limit = { has_active_mission = GEA_warlord_repay_german_loan_medium }
			set_variable = { GEA_german_loan_repayment = 100 }
		}
		else_if = {
			limit = { has_active_mission = GEA_warlord_repay_german_loan_long }
			set_variable = { GEA_german_loan_repayment = 200 }
		}
	}
}
