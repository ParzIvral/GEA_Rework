# Called in the Chinese country's scope. Keep all regional counters in GEA.
# The current Nanjing owner uses the same LEP-region counters as investment from GEA.

		limit = { OR = { owns_state = 613 original_tag = LEP original_tag = ANQ } }

			add_to_variable = { GEA_china_lep_attitude = $AMOUNT$ }
