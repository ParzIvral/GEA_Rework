GEA_Indochina_click_effect = {
	if = {
		limit = {
			NOT = {
				has_country_flag = Indochina_OPEN
			}
		}
		set_country_flag = Indochina_OPEN
		clr_country_flag = Malaya_OPEN
		clr_country_flag = Borneo_OPEN
		clr_country_flag = Guinea_OPEN
		clr_country_flag = Samoa_OPEN
		clr_country_flag = Shandong_OPEN
		clr_country_flag = Solomon_OPEN
	}
	
	else_if = {
		limit = {
			has_country_flag = Indochina_OPEN
		}
		clr_country_flag = Indochina_OPEN
	}
}
